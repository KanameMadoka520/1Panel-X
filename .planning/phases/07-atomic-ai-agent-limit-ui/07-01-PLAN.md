1. **`agent/app/service/agents.go`** — add a package-level limiter:
   ```go
   type aiAgentLimiter struct {
       mu       sync.Mutex
       inFlight int64
   }
   var aiAgentLimiterInst = &aiAgentLimiter{}

   // reserve reads the committed count under the lock and reserves one slot if
   // count+inFlight < limit. Returns a release that decrements inFlight once.
   func (l *aiAgentLimiter) reserve(limit int64, countFn func() (int64, error)) (func(), error) {
       if limit <= 0 { return func() {}, nil }
       l.mu.Lock()
       defer l.mu.Unlock()
       count, err := countFn()
       if err != nil { return nil, err }
       if err := validateAIAgentCapacity(count+l.inFlight, limit); err != nil { return nil, err }
       l.inFlight++
       var once sync.Once
       return func() {
           once.Do(func() {
               l.mu.Lock()
               defer l.mu.Unlock()
               if l.inFlight > 0 { l.inFlight-- }
           })
       }, nil
   }

   func reserveAIAgentSlot() (func(), error) {
       return aiAgentLimiterInst.reserve(loadAIAgentLimit(), func() (int64, error) {
           count, _, err := agentRepo.Page(1, 1)
           return count, err
       })
   }
   ```
